# --- raw pipe (lightest mode, default ON) ---
# video bytes are copied straight from the CDN to the browser, no web framework in between.

# RAW_PIPE=1

# --- CPU: find the real cause ---
# While a video plays, open   https://<your-wispbyte>/_prof?k=<LINK_SECRET>&s=20   (20 s profile, text) and send it.
# The logs also get a line every 10 s:  [cpu] ... cpu=..% requests=.. data=..MB new_tls=.. reused=..
# POOL_TTL=20     seconds an idle CDN connection is kept for the next request
# POOL_MAX=4      idle connections kept per CDN host
# RAW_VERIFY_TLS=0  skips the CDN certificate check (saves CPU on every new TLS connection; video bytes are not secret). Default 1.

# --- HTTP FIRST (new): try plain http:// to the CDN, so the relay needs no TLS decrypt (biggest CPU cost) ---
# HTTP_FIRST=1   (default) 1 = try http first, 0 = always https (old behaviour)
# HTTP_TTL=86400 seconds a test result is remembered per CDN host (24 h). works -> http for 24 h, refused -> https for 24 h,
#                then ONE new test. One failed http request while http is marked "works" is repeated over https for that
#                request only. The result is lost when the relay restarts (first request tests again).
# Logs: a line "[relay] http test <host>: WORKS / refused" at each test, and http_pipe=<n> in the [cpu] line.

# --- TLS PASS-THROUGH (new): relay does NOT decrypt the CDN's https, the phone does (needs sw.js + tlsdec.js on the site) ---
# The phone's service worker asks  /pt?...  instead of  /p?...  The relay then does only the TLS handshake and the request (with the swapped
# Referer), and copies the locked bytes to the phone together with the key of that one connection (header X-Tls). Order of tries per request:
#   1. http://  (HTTP_FIRST, remembered 24 h)   2. pass-through (this)   3. normal https with decrypt on the relay (old way)
# TLS_PASS=1          (default) 0 = never pass through
# PASS_BAD_TTL=1800   seconds a CDN host is skipped after pass-through was not possible for it (no TLS 1.2 / no AES-GCM / handshake error)
# PASS_SESSION_TTL=3600  seconds a TLS session is reused (a resumed handshake costs about 1/4 of a full one)
# PASS_KEYLOG=        (not needed) the relay writes the session keys of its own CDN connections to /dev/shm (or the temp dir) with mode 600
#                     and empties that file when it passes 256 KB. Nothing else is stored.
# Logs:  "[relay] tls pass-through <host>: WORKS ..."  or  "not possible (<reason>)"  or  "answer is <status> ..."
#        "[relay] http test <host>: refused - https; reason: <status / Location / error>"   <- why http:// was refused
#        [cpu] line: tls_pass=<n> = requests that were passed through
